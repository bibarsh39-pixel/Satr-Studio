cmake_minimum_required(VERSION 3.16)
project(SatrStudio VERSION 0.9.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(wxWidgets 3.2 REQUIRED COMPONENTS core base aui stc xml)
include(${wxWidgets_USE_FILE})
add_executable(Satr src/main.cpp src/diagnostics.cpp)
target_include_directories(Satr PRIVATE third_party)
target_link_libraries(Satr PRIVATE ${wxWidgets_LIBRARIES})
target_compile_options(Satr PRIVATE -Wall -Wextra)
if(UNIX AND NOT APPLE)
    add_executable(SatrRunner src/console_runner_posix.cpp)
    target_include_directories(SatrRunner PRIVATE third_party)
else()
    message(FATAL_ERROR "Use build.ps1 for the Windows release")
endif()
